cmake_minimum_required(VERSION 3.21)
project(pbrt4_consumer LANGUAGES CXX)
if(PBRT4_SOURCE)
  if(PBRT4_FETCHCONTENT)
    include(FetchContent)
    FetchContent_Declare(pbrt4 SOURCE_DIR "${PBRT4_SOURCE}")
    FetchContent_MakeAvailable(pbrt4)
  else()
    add_subdirectory("${PBRT4_SOURCE}" sdk)
  endif()
else()
  find_package(pbrt4 0.1 CONFIG REQUIRED)
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE pbrt4::pbrt4)
if(TARGET pbrt4::geometry)
  target_link_libraries(consumer PRIVATE pbrt4::geometry)
  target_compile_definitions(consumer PRIVATE HAS_GEOMETRY=1)
endif()
enable_testing()
add_test(NAME consumer_runs COMMAND consumer)
