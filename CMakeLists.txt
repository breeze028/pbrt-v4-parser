cmake_minimum_required(VERSION 3.20)
project(pbrt4 VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(PBRT4_BUILD_TESTS "Build SDK tests" ${PROJECT_IS_TOP_LEVEL})
option(PBRT4_BUILD_TOOLS "Build command line tools" ${PROJECT_IS_TOP_LEVEL})
option(PBRT4_BUILD_EXAMPLES "Build examples" ${PROJECT_IS_TOP_LEVEL})
add_library(pbrt4 src/parser/parser.cpp src/scene/math.cpp src/scene/builder.cpp)
add_library(pbrt4::pbrt4 ALIAS pbrt4)
target_compile_features(pbrt4 PUBLIC cxx_std_20)
set_target_properties(pbrt4 PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(pbrt4 PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(MSVC)
  target_compile_options(pbrt4 PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(pbrt4 PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(PBRT4_BUILD_TESTS)
  enable_testing()
  add_executable(pbrt4_tests tests/tests.cpp)
  target_link_libraries(pbrt4_tests PRIVATE pbrt4::pbrt4)
  target_compile_definitions(pbrt4_tests PRIVATE PBRT4_TESTDATA="${CMAKE_CURRENT_SOURCE_DIR}/testdata")
  add_test(NAME pbrt4_tests COMMAND pbrt4_tests)
endif()
install(TARGETS pbrt4 EXPORT pbrt4Targets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT pbrt4Targets NAMESPACE pbrt4:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/pbrt4)
configure_package_config_file(cmake/pbrt4Config.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/pbrt4Config.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/pbrt4)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/pbrt4ConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/pbrt4Config.cmake ${CMAKE_CURRENT_BINARY_DIR}/pbrt4ConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/pbrt4)
