cmake_minimum_required(VERSION 3.21)
project(pbrt4 VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(PBRT4_BUILD_TESTS "Build SDK tests" ${PROJECT_IS_TOP_LEVEL})
option(PBRT4_BUILD_TOOLS "Build command line tools" ${PROJECT_IS_TOP_LEVEL})
option(PBRT4_BUILD_EXAMPLES "Build examples" ${PROJECT_IS_TOP_LEVEL})
option(PBRT4_BUILD_GEOMETRY "Build optional geometry utilities" ON)
option(PBRT4_TEST_LARGE_SCENES "Test locally copied bathroom/spaceship assets" OFF)
add_library(pbrt4 src/parser/parser.cpp src/scene/math.cpp src/scene/builder.cpp)
add_library(pbrt4::pbrt4 ALIAS pbrt4)
target_compile_features(pbrt4 PUBLIC cxx_std_20)
set_target_properties(pbrt4 PROPERTIES CXX_EXTENSIONS OFF WINDOWS_EXPORT_ALL_SYMBOLS ON POSITION_INDEPENDENT_CODE ON)
target_include_directories(pbrt4 PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(MSVC)
  target_compile_options(pbrt4 PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(pbrt4 PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(PBRT4_BUILD_GEOMETRY)
  add_library(pbrt4_geometry src/geometry/geometry.cpp)
  add_library(pbrt4::geometry ALIAS pbrt4_geometry)
  set_target_properties(pbrt4_geometry PROPERTIES EXPORT_NAME geometry CXX_EXTENSIONS OFF WINDOWS_EXPORT_ALL_SYMBOLS ON POSITION_INDEPENDENT_CODE ON)
  target_link_libraries(pbrt4_geometry PUBLIC pbrt4::pbrt4)
  if(MSVC)
    target_compile_options(pbrt4_geometry PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(pbrt4_geometry PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  install(TARGETS pbrt4_geometry EXPORT pbrt4Targets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
if(PBRT4_BUILD_TESTS)
  enable_testing()
  add_executable(pbrt4_tests tests/core_tests.cpp)
  find_package(Threads REQUIRED)
  target_link_libraries(pbrt4_tests PRIVATE pbrt4::pbrt4 Threads::Threads)
  target_compile_definitions(pbrt4_tests PRIVATE PBRT4_TESTDATA="${CMAKE_CURRENT_SOURCE_DIR}/testdata")
  foreach(case tokenizer parameters all_directives syntax_errors lexical_errors unicode_paths transform camera attributes animation instances references include import import_scope_errors medium_spectrum semantic_errors file_errors independent_loads malformed_corpus)
    add_test(NAME ${case} COMMAND pbrt4_tests ${case})
  endforeach()
  if(PBRT4_BUILD_GEOMETRY)
    add_executable(pbrt4_geometry_tests tests/geometry_tests.cpp)
    target_link_libraries(pbrt4_geometry_tests PRIVATE pbrt4::geometry)
    target_compile_definitions(pbrt4_geometry_tests PRIVATE PBRT4_TESTDATA="${CMAKE_CURRENT_SOURCE_DIR}/testdata")
    foreach(case ascii_ply binary_ply ply_errors triangle_mesh shared_geometry sphere bake_normals)
      add_test(NAME ${case} COMMAND pbrt4_geometry_tests ${case})
    endforeach()
  endif()
endif()
if(PBRT4_BUILD_TOOLS)
  add_executable(pbrt4_info tools/pbrt4_info/main.cpp)
  target_link_libraries(pbrt4_info PRIVATE pbrt4::pbrt4)
  if(PBRT4_BUILD_GEOMETRY)
    target_link_libraries(pbrt4_info PRIVATE pbrt4::geometry)
    target_compile_definitions(pbrt4_info PRIVATE PBRT4_INFO_GEOMETRY=1)
  endif()
  if(PBRT4_BUILD_TESTS)
    add_test(NAME info_resources COMMAND pbrt4_info --details --check-resources ${CMAKE_CURRENT_SOURCE_DIR}/testdata/ply-instances.pbrt)
    add_test(NAME info_missing_resources COMMAND pbrt4_info --check-resources ${CMAKE_CURRENT_SOURCE_DIR}/testdata/missing-resources.pbrt)
    set_tests_properties(info_missing_resources PROPERTIES WILL_FAIL TRUE)
    if(PBRT4_BUILD_GEOMETRY)
      add_test(NAME info_geometry COMMAND pbrt4_info --geometry ${CMAKE_CURRENT_SOURCE_DIR}/testdata/ply-instances.pbrt)
    endif()
  endif()
endif()
if(PBRT4_BUILD_EXAMPLES)
  add_executable(minimal_loader examples/minimal_loader/main.cpp)
  target_link_libraries(minimal_loader PRIVATE pbrt4::pbrt4)
  if(PBRT4_BUILD_GEOMETRY)
    add_executable(tiny_pathtracer examples/tiny_pathtracer/main.cpp)
    target_link_libraries(tiny_pathtracer PRIVATE pbrt4::geometry)
    if(PBRT4_BUILD_TESTS)
      add_test(NAME tiny_render COMMAND tiny_pathtracer ${CMAKE_CURRENT_SOURCE_DIR}/testdata/cornell.pbrt ${CMAKE_CURRENT_BINARY_DIR}/cornell-test.ppm 1)
      add_test(NAME tiny_ply_render COMMAND tiny_pathtracer ${CMAKE_CURRENT_SOURCE_DIR}/testdata/ply-instances.pbrt ${CMAKE_CURRENT_BINARY_DIR}/ply-test.ppm 1)
      add_test(NAME tiny_rejects_material COMMAND tiny_pathtracer ${CMAKE_CURRENT_SOURCE_DIR}/testdata/unsupported-material.pbrt ${CMAKE_CURRENT_BINARY_DIR}/unsupported.ppm 1)
      set_tests_properties(tiny_rejects_material PROPERTIES WILL_FAIL TRUE)
      add_test(NAME tiny_cornell_box COMMAND tiny_pathtracer ${CMAKE_CURRENT_SOURCE_DIR}/scenes/cornell-box/scene-v4.pbrt ${CMAKE_CURRENT_BINARY_DIR}/cornell-box-test.ppm 1 32)
    endif()
  endif()
endif()
if(PBRT4_TEST_LARGE_SCENES)
  if(NOT PBRT4_BUILD_TESTS OR NOT PBRT4_BUILD_TOOLS OR NOT PBRT4_BUILD_GEOMETRY)
    message(FATAL_ERROR "PBRT4_TEST_LARGE_SCENES requires tests, tools, and geometry utilities")
  endif()
  foreach(scene bathroom spaceship)
    add_test(NAME example_${scene} COMMAND pbrt4_info --details --check-resources --geometry ${CMAKE_CURRENT_SOURCE_DIR}/scenes/${scene}/scene-v4.pbrt)
    set_tests_properties(example_${scene} PROPERTIES LABELS example_scene)
  endforeach()
endif()
install(TARGETS pbrt4 EXPORT pbrt4Targets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE NOTICE DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/licenses/pbrt4)
install(EXPORT pbrt4Targets NAMESPACE pbrt4:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/pbrt4)
configure_package_config_file(cmake/pbrt4Config.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/pbrt4Config.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/pbrt4)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/pbrt4ConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/pbrt4Config.cmake ${CMAKE_CURRENT_BINARY_DIR}/pbrt4ConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/pbrt4)
